function To-Int16 {
    param([int]$Value)
    if ($Value -ge 32768) { return $Value - 65536 }
    return $Value
}

function Fix-HashtableKeys {
    param([hashtable]$h)

    $new = @{}
    foreach ($k in $h.Keys) {

        # Key opschonen: BOM, whitespace, tabs, CRLF
        $clean = $k.ToString().Trim().Replace([string][char]0xFEFF, "")

        # Als het een integer is → integer key
        if ($clean -as [int]) {
            $new[[int]$clean] = $h[$k]
        }
        else {
            # Anders als string key
            $new[$clean] = $h[$k]
        }
    }
    return $new
}

# 5. Decode helpers
###############################################################

function Decode-SGReady {
    param([int]$Enabled, [int]$In1, [int]$In2)

    if ($Enabled -eq 0 -or $Enabled -eq 32768 -or $Enabled -eq 36864) { return "Disabled" }

    $key = "${In1},${In2}"
    $modes = @{
        "0,0" = "Automatic"
        "1,0" = "Accelerated"
        "0,1" = "Standby/Defrost"
        "1,1" = "Maximum"
    }

    if ($modes.ContainsKey($key)) { return $modes[$key] }
    return "Unknown SG-Ready state ($key)"
}

function Decode-ScaledValue {
    param(
        [hashtable]$Def,
        [int]$Value
    )

    # N/A handling
    if ($Value -eq 32768 -or $Value -eq 36864) {
        return "N/A"
    }

    $v = $Value
    if ($Def.Signed) {
        $v = To-Int16 $Value
    }

    $scaled = $v * $Def.Scale
    if ($Def.Unit -ne "") {
        return "$scaled $($Def.Unit)"
    } else {
        return "$scaled"
    }
}

function Decode-Register {
    param(
        [int]$Reg,
        [int]$Value
    )

    $maps = @(
        $ProcessData,
        $SystemParameters,
        $Status2500,
        $HardwareConfig,
        $Block3500,
        $SGReady
    )

    foreach ($map in $maps) {
        if ($map.ContainsKey($Reg)) {
            $d = $map[$Reg]

            # N/A afhandeling
            if ($Value -eq 32768 -or $Value -eq 36864) {
                return "$($d.Name) = N/A"
            }

            # Bitfield
            if ($d.ContainsKey('Type') -and $d.Type -eq 'bitfield') {
                $bits = $d.Bits
                $decodedBits = @()
                foreach ($bit in $bits.Keys) {
                    if ($Value -band (1 -shl $bit)) {
                        $decodedBits += $bits[$bit]
                    }
                }
                if ($decodedBits.Count -eq 0) {
                    return "$($d.Name) = None"
                }
                return "$($d.Name) = " + ($decodedBits -join ', ')
            }

            # Enum
            if ($d.ContainsKey('Type') -and $d.Type -eq 'enum') {
                $vals = $d.Values
                if ($vals.ContainsKey($Value)) {
                    return "$($d.Name) = $($vals[$Value])"
                } else {
                    return "$($d.Name) = Unknown ($Value)"
                }
            }

            # Gewone geschaalde waarde
            $display = Decode-ScaledValue -Def $d -Value $Value
            return "$($d.Name) = $display"
        }
    }

    if ($Value -eq 32768 -or $Value -eq 36864) {
        return "Unknown register ${Reg} = N/A"
    }

    return "Unknown register ${Reg} = ${Value}"
}

###############################################################
# 7. Dump all registers (collect into objects for GridView)
###############################################################

$results = @()

foreach ($range in $Ranges) {

    foreach ($reg in $range) {

        $isInput = ($reg -ge 500 -and $reg -le 600) -or
                   ($reg -ge 2500 -and $reg -le 2600) -or
                   ($reg -ge 3500 -and $reg -le 3600)

        try {
            if ($isInput) {
                $val = $master.ReadInputRegisters($slave, $reg, 1)[0]
            } else {
                $val = $master.ReadHoldingRegisters($slave, $reg, 1)[0]
            }
        }
        catch {
            $results += [PSCustomObject]@{
                Register    = $reg
                RawValue    = "ERROR"
                ScaledValue = "N/A"
                Decoded     = "Illegal address or unsupported"
            }
            continue
        }

        # Decode full text
        $decoded = Decode-Register -Reg $reg -Value $val

        # Determine scaled value (if mapping exists)
        $scaled = $null
        $maps = @($ProcessData, $SystemParameters, $Status2500, $HardwareConfig, $Block3500, $SGReady)

        $found = $false
        foreach ($map in $maps) {
            if ($map.ContainsKey($reg)) {
                $d = $map[$reg]

                # N/A
                if ($val -eq 32768 -or $val -eq 36864) {
                    $scaled = "N/A"
                }
                # Bitfield → raw integer
                elseif ($d.ContainsKey('Type') -and $d.Type -eq 'bitfield') {
                    $scaled = $val
                }
                # Enum → tekst
                elseif ($d.ContainsKey('Type') -and $d.Type -eq 'enum') {
                    $key = [int]$val
                    if ($d.Values.ContainsKey($key)) {
                        $scaled = $d.Values[$key]
                    } else {
                        $scaled = "Unknown ($val)"
                    }
                }

                # Scalar → geschaalde waarde
                else {
                    $scaled = Decode-ScaledValue -Def $d -Value $val
                }

                $found = $true
                break
            }
        }

        if (-not $found) {
            if ($val -eq 32768 -or $val -eq 36864) {
                $scaled = "N/A"
            } else {
                $scaled = $val
            }
        }

        # Add row to results
        $results += [PSCustomObject]@{
            Register    = $reg
            RawValue    = $val
            ScaledValue = $scaled
            Decoded     = $decoded
        }
    }
}

# Voeg SG-Ready gecombineerde status toe als virtuele rij
$sg0 = $master.ReadHoldingRegisters($slave, 4000, 1)[0]
$sg1 = $master.ReadHoldingRegisters($slave, 4001, 1)[0]
$sg2 = $master.ReadHoldingRegisters($slave, 4002, 1)[0]

$results += [PSCustomObject]@{
    Register    = "SGR"
    RawValue    = "$sg0,$sg1,$sg2"
    ScaledValue = $sgDecoded
    Decoded     = "SG-Ready Mode = $sgDecoded"
}
