function Get-ISGMapping {
    $all = @{}
    foreach ($map in @(
        $script:ProcessData,
        $script:SystemParameters,
        $script:Status,
        $script:HardwareConfig,
        $script:SGReady,
        $script:Block3500
    )) {
        foreach ($k in $map.Keys) {
            $all[$k] = $map[$k]
        }
    }
    return $all
}
